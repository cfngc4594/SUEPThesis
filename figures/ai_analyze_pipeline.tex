\begin{tikzpicture}[
  font=\scriptsize\sffamily,
  st/.style={draw, rounded corners=2pt, align=center, inner sep=4pt,
             minimum width=2.0cm, minimum height=0.7cm},
  st1/.style={st, fill=blue!8},
  st2/.style={st, fill=green!10},
  st3/.style={st, fill=orange!10},
  st4/.style={st, fill=purple!10},
  st5/.style={st, fill=red!10},
  ext/.style={draw, dashed, rounded corners=2pt, align=center, inner sep=4pt,

  arr/.style={-{Stealth[length=2mm]}, thick},
  lab/.style={font=\tiny, fill=white, inner sep=1pt}
]
  % 主状态线（横向，间距加大）
  \node[st1] (Q)   {QUEUED};
  \node[st2, right=1.2cm of Q]  (P) {PROCESSING};

  \node[st5, below=1.6cm of $(P)!0.5!(T)$] (F) {FAILED};

  % 起点：判题完成后触发（左侧外部触发，与 Q 留出间距）
  \node[ext, left=1.0cm of Q, fill=gray!10] (J)

  \node[ext, above=1.6cm of T, fill=cyan!10] (M)
        {\texttt{generateText} +\\\texttt{toolChoice: saveCodeAnalysis}};
  \node[ext, below=1.6cm of C, fill=yellow!12] (DB)
        {\texttt{CodeAnalysis} 表\\结构化字段落库};

  % 主迁移
  \draw[arr] (J) -- node[lab, above]{创建记录} (Q);
  \draw[arr] (Q) -- node[lab, above]{开始处理} (P);
  \draw[arr] (P) -- node[lab, above]{发起调用} (T);
  \draw[arr] (T) -- node[lab, above]{工具回调 execute} (C);

  % 异常分支：分别从 P 和 T 走斜线到 FAILED，避免重叠
  \draw[arr] (P.south) -- node[lab, sloped, above]{异常} (F.north west);
  \draw[arr] (T.south) -- node[lab, sloped, above]{解析失败} (F.north east);

  % 与模型双向交互：去程实线左侧，回程虚线右侧
  \draw[arr]         ([xshift=-7pt]T.north) -- node[lab, left]{Chat Completions}
                                                ([xshift=-7pt]M.south);
  \draw[arr, dashed] ([xshift= 7pt]M.south)  -- node[lab, right]{结构化字段}
                                                ([xshift= 7pt]T.north);

  % 与数据库交互
  \draw[arr]         (C) -- node[lab, right]{UPDATE} (DB);
  \draw[arr, dashed] (F.east) to[bend right=20]
                     node[lab, below, pos=0.55]{标记 FAILED} (DB.west);
